-- Keeps track of the items in a warehouse.

CREATE TABLE warehouse (
    id      SERIAL PRIMARY KEY,
    name    VARCHAR(100) NOT NULL UNIQUE,
    opened  DATE DEFAULT CURRENT_DATE
);

CREATE TABLE item (
    id            BIGINT PRIMARY KEY,
    warehouse_id  INT NOT NULL REFERENCES warehouse (id) ON DELETE CASCADE,
    name          VARCHAR(200) NOT NULL,
    quantity      INT DEFAULT 0 CHECK (quantity >= 0),
    price         DECIMAL(10, 2),
    fragile       BOOLEAN DEFAULT FALSE,
    notes         TEXT
);

CREATE INDEX item_name ON item (name);

-- total value per warehouse, tax included
SELECT w.name,
       COUNT(*)                                AS items,
       SUM(i.quantity * i.price * (1 + 0.075)) AS total_value
FROM warehouse w
         INNER JOIN item i ON i.warehouse_id = w.id
WHERE i.quantity > 0
  AND i.notes IS NULL
  AND w.opened BETWEEN :since AND :until
GROUP BY w.name
HAVING COUNT(*) >= 2
ORDER BY total_value DESC
LIMIT 10;

UPDATE item SET quantity = quantity - 1 WHERE id = @itemId AND fragile = FALSE;
